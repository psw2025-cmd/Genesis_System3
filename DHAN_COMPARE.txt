_ACCOUNT_CACHE: Dict[str, Any] = {}
_ACCOUNT_CACHE_TTL_S = 20.0

def _cached_call(key: str, fn):
    now = time.monotonic()
    entry = _ACCOUNT_CACHE.get(key)
    if entry and (now - entry[0]) < _ACCOUNT_CACHE_TTL_S:
        return entry[1]
    val = fn()
    _ACCOUNT_CACHE[key] = (now, val)
    return val


def _unwrap_dhan_data(raw: Any, default: Any) -> Any:
    if not isinstance(raw, dict):
        return raw if isinstance(raw, type(default)) else default
    data = raw.get("data")
    if isinstance(data, dict) and "data" in data and ("status" in data or "remarks" in data):
        data = data.get("data")
    return data if data is not None else default


def _normalize_holding(h: Dict[str, Any]) -> Dict[str, Any]:
    qty = float(h.get("totalQty", h.get("quantity", 0)) or 0)
    avg_cost = float(h.get("avgCostPrice", h.get("avg_cost_price", 0.0)) or 0.0)
    ltp = float(h.get("lastTradedPrice", h.get("last_traded_price", 0.0)) or 0.0)
    invested = round(qty * avg_cost, 2)
    current = round(qty * ltp, 2)
    pnl = round(current - invested, 2)
    pnl_pct = round((pnl / invested * 100.0), 2) if invested > 0 else 0.0
    out = dict(h)
    out.update({
        "symbol": h.get("tradingSymbol") or h.get("symbol") or "",
        "security_id": str(h.get("securityId") or h.get("security_id") or ""),
        "isin": h.get("isin") or "",
        "quantity": int(qty) if qty.is_integer() else qty,
        "avg_cost_price": round(avg_cost, 2),
        "last_traded_price": round(ltp, 2),
        "invested_value": invested,
        "current_value": current,
        "unrealized_pnl": pnl,
        "unrealized_pnl_pct": pnl_pct,
    })
    return out

def _session(now: datetime) -> str:
    local = now.astimezone(IST)
    if local.weekday() > 4 or local.time() < day_time(9, 15) or local.time() >= day_time(15, 30):
        return "CLOSED"
    return "OPEN"


def _ids():
    version = MASTER.stat().st_mtime_ns
    existing = globals().get("_market_id_cache")
    if existing and existing[0] == version:
        return existing[1]
    found = {}
    with MASTER.open(encoding="utf-8-sig", newline="") as f:
        for r in csv.DictReader(f):
            if r.get("INSTRUMENT") != "INDEX" or r.get("SEGMENT") != "I":
                continue
            name = (r.get("SYMBOL_NAME") or "").upper()
            if name in NAMES:
                ex = r.get("EXCH_ID")
                if (name == "SENSEX" and ex == "BSE") or (name != "SENSEX" and ex == "NSE"):
                    try:
                        found[name] = int(r["SECURITY_ID"])
                    except (ValueError, TypeError):
                        pass
    globals()["_market_id_cache"] = (version, found)
    return found

def _parse_trade_time(value):
    if not value:
        return None
    try:
        d = datetime.strptime(str(value), "%d/%m/%Y %H:%M:%S").replace(tzinfo=IST)
        return d if d.year >= 2020 else None
    except (ValueError, TypeError):
        return None

def _fetch():
    now = datetime.now(timezone.utc)
    if not MASTER.is_file() or not MANIFEST.is_file():
        return _error("MASTER_MISSING", now, "Fresh Dhan instrument master unavailable")
    try:
        meta = json.loads(MANIFEST.read_text(encoding="utf-8"))
        downloaded = datetime.fromisoformat(meta["downloaded_at_utc"].replace("Z", "+00:00"))
        if (now - downloaded).total_seconds() > 36 * 3600:
            return _error("MASTER_STALE", now, "Instrument master exceeds 36 hours", meta)
        token = ""
        client = ""
        sec_p = Path(r"C:\Genesis_System3_clean\.secrets\dhan.env")
        if sec_p.is_file():
            for line in sec_p.read_text(encoding="utf-8-sig").splitlines():
                s = line.strip()
                if not s or s.startswith("#") or "=" not in s:
                    continue
                k, v = s.split("=", 1)
                k, v = k.strip(), v.strip().strip('"').strip("'")
                if k == "DHAN_ACCESS_TOKEN":
                    token = v
                elif k == "DHAN_CLIENT_ID":
                    client = v
        if not token:
            token = os.environ.get("DHAN_ACCESS_TOKEN", "").strip()
        if not client:
            client = os.environ.get("DHAN_CLIENT_ID", "").strip()
        ids = _ids()
        if not token or not client:
            return _get_eod_fallback(meta, ids, "Credentials missing in local env; serving last verified Dhan quote snapshot")
    except (ValueError, OSError, KeyError, TypeError):
        return _error("MASTER_INVALID", now, "Instrument master cannot be verified")
    if set(ids) != set(NAMES):
        return _error("MASTER_INCOMPLETE", now, "Index security IDs could not all be resolved", meta, ids)
    body = json.dumps({"IDX_I": [ids[s] for s in NAMES]}).encode("utf-8")
    req = urllib.request.Request(
        MARKET_URL,
        data=body,
        method="POST",
        headers={
            "access-token": token,
            "client-id": client,
            "Accept": "application/json",
            "Content-Type": "application/json",
        },
    )
    try:
        with urllib.request.urlopen(req, timeout=4) as resp:
            raw = resp.read(512 * 1024)
            if not raw:
                raise ValueError("empty provider response")
            payload = json.loads(raw)
        if payload.get("status") != "success":
            return _get_eod_fallback(meta, ids)
        items = payload.get("data", {}).get("IDX_I", {})
        output = {}
        for name, security_id in ids.items():
            q = items.get(str(security_id), {})
            try:
                price = float(q.get("last_price"))
            except (ValueError, TypeError):
                price = 0.0
            stamp = _parse_trade_time(q.get("last_trade_time"))
            if price <= 0 or not stamp or (stamp - now.astimezone(IST)).total_seconds() > 300:
                output[name] = {"status": "NO_VERIFIED_TIMESTAMP", "price": None}
                continue
            age = max(0.0, (now - stamp.astimezone(timezone.utc)).total_seconds())
            if age > 24 * 3600:
                output[name] = {"status": "STALE", "price": None, "last_trade_ist": stamp.isoformat()}
            else:
                state = "LAST_TRADE" if _session(now) == "CLOSED" else "QUOTE_WITH_EXCHANGE_TIME"
                prev_c = OFFICIAL_PREV_CLOSE.get(name, price)
                chg = round(price - prev_c, 2)
                chg_pct = round(((price / prev_c) - 1.0) * 100, 2) if prev_c > 0 else 0.0
                output[name] = {
                    "status": state,
                    "price": price,
                    "previous_close": prev_c,
                    "change": chg,
                    "change_pct": chg_pct,
                    "last_trade_ist": stamp.isoformat(),
                    "age_seconds": round(age, 1),
                    "security_id": security_id,
                    "segment": "IDX_I",
                }
        valid = sum(isinstance(x.get("price"), float) for x in output.values())
        if valid == 0:
            return _get_eod_fallback(meta, ids)
        res = {
            "state": "BROKER_QUOTES_RECEIVED" if valid else "PROVIDER_TIME_UNVERIFIED",
            "reason": "Dhan authenticated market-quote response; no order requests sent",
            "checked_at_utc": now.isoformat(),
            "market_session": _session(now),
            "quotes": output,
            "exchange_quotes_verified": valid > 0,
            "verified_instruments": valid,
            "read_only": True,
            "paper_mode": True,
            "live_orders_allowed": False,
            "instrument_master": meta,
            "resolved_indices": ids,
            "data_url": "https://dhanhq.co/docs/v2/market-quote/",
        }
        try:
            LAST_LIVE_CACHE.parent.mkdir(parents=True, exist_ok=True)
            LAST_LIVE_CACHE.write_text(json.dumps(res, indent=2), encoding="utf-8")
        except Exception:
            pass
        return res
    except Exception:
        return _get_eod_fallback(meta, ids)


@router.get("/api/verified-market", tags=["Provenance"])
def verified_market():
    with _mutex:
        if time.monotonic() < _cache["until"] and _cache["result"] is not None:
            return _cache["result"]
        try:
            result = _fetch()
            if not result.get("quotes"):
                meta = json.loads(MANIFEST.read_text(encoding="utf-8")) if MANIFEST.is_file() else None
                result = _get_eod_fallback(meta, _ids() if MASTER.is_file() else None)
        except Exception:
            meta = json.loads(MANIFEST.read_text(encoding="utf-8")) if MANIFEST.is_file() else None
            result = _get_eod_fallback(meta, _ids() if MASTER.is_file() else None)
        _cache["result"] = result
        _cache["until"] = time.monotonic() + 12.0
        return result

    _state_store = state_store
    _token_manager = token_manager


def _safe_error(exc: Exception, limit: int = 120) -> str:
    return str(exc)[:limit]


def _extract_payload(resp: Dict[str, Any], key: str) -> list | dict:
    """Normalize read-only adapter response payloads."""
    if not isinstance(resp, dict):
        return []
    data = resp.get("data")
    if isinstance(data, dict) and key in data:
        return data.get(key) or []
    if isinstance(data, list):
        return data
    if isinstance(data, dict):
        nested = data.get("data")
        if isinstance(nested, list):
            return nested
    return []


def _state_broker_truth(status: Dict[str, Any]) -> Dict[str, Any]:
    """Allow-list the current broker status before persisting it into SSOT."""
    connected = bool(status.get("connected", False))
    truth: Dict[str, Any] = {
        "connected": connected,
        "broker": "dhan",
        "name": "dhan",
        "status": "connected" if connected else "disconnected",
        "error": None if connected else (status.get("error") or "BROKER_NOT_CONNECTED"),
        "latency_ms": status.get("latency_ms"),
        "truth_source": "dhan_readonly_probe",
        "observed_at_utc": datetime.now(timezone.utc).isoformat(),
    }
    for key in ("auth_classification", "upstream_classification", "upstream_code"):
        if status.get(key) is not None:
            truth[key] = status.get(key)
    return truth


def _persist_broker_truth(status: Dict[str, Any]) -> None:
    """Persist both success and failure so `/api/state` cannot stay stale green."""
    if _state_store is None or not hasattr(_state_store, "update_state"):
        return
    try:
        _state_store.update_state({"broker": _state_broker_truth(status)})
    except Exception:
        # Status reads must stay available even if the optional SSOT persistence
        # layer is temporarily unavailable.  The response itself remains current.
        return

"""Pure helpers: attach authenticated snapshot spots to gain-rank rows.

Never invent BASE_SPOT / synthetic prices. Only apply spots from caller-supplied
authenticated lookup (live chain or MARKET_CLOSED_DHAN_SNAPSHOT buffers).
"""

from __future__ import annotations

from typing import Any, Dict, List, Mapping, MutableMapping, Optional


_BLOCKED_SOURCES = frozenset(
    {
        "synthetic",
        "base_spot",
        "base_spot_prices",
        "invented",
        "dummy",
    }
)


def _normalize_source(source: Any) -> str:
    return str(source or "").strip().lower()


def is_authenticated_spot_entry(entry: Mapping[str, Any] | None) -> bool:
    if not isinstance(entry, Mapping):
        return False
    try:
        spot = float(entry.get("spot") or 0)
    except (TypeError, ValueError):
        return False
    if spot <= 0:
        return False
    source = _normalize_source(entry.get("source") or entry.get("data_source"))
    if source in _BLOCKED_SOURCES:
        return False
    status = str(entry.get("status") or "").upper()
    # Explicit synthetic markers
    if "SYNTHETIC" in status:
        return False
    return True


def enrich_gain_rank_rows_with_authenticated_spots(
    rows: List[MutableMapping[str, Any]] | List[Dict[str, Any]],
    spot_lookup: Mapping[str, Mapping[str, Any]],
) -> List[Dict[str, Any]]:
    """Copy rows and fill missing spot_price from authenticated lookup only."""
    out: List[Dict[str, Any]] = []
    for row in rows or []:
        if not isinstance(row, dict):
            continue
        enriched = dict(row)
        try:
            existing = float(enriched.get("spot_price") or 0)
        except (TypeError, ValueError):
            existing = 0.0
        if existing > 0:
            out.append(enriched)
            continue
        underlying = str(enriched.get("underlying") or enriched.get("symbol") or "").upper()
        entry = spot_lookup.get(underlying) if underlying else None
        if not is_authenticated_spot_entry(entry):
            out.append(enriched)
            continue
        assert entry is not None
        enriched["spot_price"] = float(entry["spot"])
        enriched["spot_price_source"] = str(
            entry.get("source") or entry.get("data_source") or entry.get("status") or "AUTHENTICATED_SNAPSHOT"
        )
        out.append(enriched)
    return out
